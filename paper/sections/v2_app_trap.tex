\section{Model F: proofs and remarks}\label{app:trap}

\paragraph{Model F.} Fix $K\in\mathbb N$, costs $c_1,\dots,c_K>0$ (all-miss GPU seconds of turn $k$), and
$1=G_1\ge G_2\ge\dots\ge G_K>0$ with $G_{K+1}=0$ ($G_k=P(T\ge k)$). Write
\begin{gather*}
p_k=G_k-G_{k+1},\qquad r_k=G_{k+1}/G_k,\\
a_k=\lambda c_kG_k,\qquad \rho=\textstyle\sum_ka_k=\lambda/\lmiss,
\end{gather*}
with $\lmiss=1/\sum_kc_kG_k$. A session whose next turn is $k$ has remaining work $R_k=\sum_{j\ge k}c_jG_j/G_k$, so that
$R_1=1/\lmiss$ and $R_k=c_k+r_kR_{k+1}$. The model assumes:
\begin{enumerate}[(F1),leftmargin=*,nosep]
\item every turn misses, so turn $k$ costs $c_k$;
\item the server does one GPU second of work per second while work is pending;
\item requests complete in the order in which they joined;
\item a completed turn $k<T$ rejoins at once as turn $k+1$, so a fraction $r_k$ of the class-$k$ mass completing at any time
rejoins as class $k+1$;
\item new sessions join as class-1 mass at rate $\lambda$;
\item at time 0 a pending mass $m^0_k\ge0$ of each class is served first, with $V_0=\sum_kc_km^0_k>0$, through absolutely
continuous class-completion functions $C^0_k$ on $[0,V_0]$ with $\sum_kc_kC^0_k(t)=t$.
\end{enumerate}
A \emph{solution} consists of non-decreasing absolutely continuous $J_k$, $C_k$ (joins and completions after time 0, with
$C^0_k(0)=0$) on $[0,\tau)$ such that
\begin{itemize}[leftmargin=*,nosep]
\item $J_1(t)=\lambda t$ and $J_{k+1}=r_kC_k$;
\item $C_k=C^0_k$ on $[0,V_0]$, and $C_k(f(u))=m^0_k+J_k(u)$ for $u\ge0$ (FIFO);
\item $\sum_kc_kC_k(t)=t$ for $t<\tau$ (F2);
\item the pending work $V(u)=\sum_kc_k\pi_k(u)$ is positive on $[0,\tau)$;
\item either $\tau=\infty$ or $V(t)\to0$ as $t\uparrow\tau$.
\end{itemize}
Here $f(u)=u+V(u)$ and $\pi_k=m^0_k+J_k-C_k$. The last condition makes $\tau\le\infty$ the time at which the queue first
empties, rather than any earlier cut-off. With zero think time every
session present has one pending request, so the number in the system is $N=\sum_k\pi_k$. The remaining work is
$V_{\mathrm{tot}}=\sum_k\pi_kR_k$, and the departures are $\mathrm{Dep}(t)=\sum_km^0_k+\lambda t-N(t)$.

On the engine, (F1) is the measured zero hit fraction of the trap and (F2) its saturated GPU. (F3) is request-level FCFS
admission (Proposition~\ref{prop:A}), up to the $M/E_S\approx28$ requests that run concurrently. For (F4), the 5\,s tool
time is short against cycles of 166--516\,s, and including it changes the window statistics of the fluid by less than
$10^{-3}$ (\fpath{analysis/theory/icml/support.py}). The costs are $c_k=t^m_k+\delta$: the calibrated miss prefill of
\S\ref{sec:fp-model} plus the all-miss per-turn decode share $\delta$, chosen so that $\sum_k(G_k/E[T])c_k=1/\nu_b$. Then
$\lmiss$ above equals the model's $\mu_b$ (Proposition~\ref{prop:nb}(a)).

\begin{lemma}[Perron root]\label{lem:perron}
$\varphi(x)=\sum_ka_kx^{-k}$ decreases strictly from $\infty$ to $0$ on $(0,\infty)$, and $\varphi(1)=\rho$. Hence
$\varphi(z)=1$ has a unique root $z>0$, with $z>1$ if and only if $\rho>1$.
\end{lemma}

\begin{lemma}[well-posedness, work, orbit recursion]\label{lem:orbit}
Under (F1)--(F6):
\begin{enumerate}[(i),leftmargin=*,nosep]
\item model F has exactly one solution;
\item while the server is busy, $f(u)=V_0+W(u)$ with $W=\sum_kc_kJ_k$, and $f$ is continuous and strictly increasing;
\item while busy, $V_{\mathrm{tot}}(t)=V_{\mathrm{tot}}(0)+(\rho-1)t$.
\end{enumerate}
\begin{enumerate}[(i),leftmargin=*,nosep,resume]
\item For $u_0\in[0,V_0)$, let $u_{n+1}=f(u_n)$ and $\Delta_n=u_{n+1}-u_n$. Then
\begin{gather*}
\Delta_n=\sum_{k=1}^{\min(n,K)}a_k\Delta_{n-k}+h_n(u_0),\\
h_n(u_0)=\sum_{k=n+1}^{K}c_k\frac{G_k}{G_{k-n}}\pi_{k-n}(u_0),
\end{gather*}
so that $h_0=V(u_0)$ and $h_n=0$ for $n\ge K$. Each $h_n$ is continuous in $u_0$, and $\sum_nh_n(u_0)=V_{\mathrm{tot}}(u_0)$.
\item For $n\ge1$, $\pi_k(u_n)=\lambda G_k\Delta_{n-k}$ if $k\le n$, and $\pi_k(u_n)=(G_k/G_{k-n})\pi_{k-n}(u_0)$ otherwise.
The requests completed in $(u_n,u_{n+1}]$ are those pending at $u_n$.
\end{enumerate}
\end{lemma}

\begin{proof}
\emph{(ii).} If the server is busy on $[0,f(u)]$, then by (F2) it has completed $s$ units of work by each time $s$ in that
interval. By (F3), the request joining at $u$ completes once the work joined before it, $V_0+W(u)$, is done. So
$f(u)=V_0+W(u)$, which is continuous. Since $W'\ge\lambda c_1>0$ almost everywhere, $f$ is strictly increasing, and
$f(0)=V_0$.

\emph{(i).} Put $b_0=0$, $b_1=V_0$ and $b_{n+1}=f(b_n)$. On $[0,b_1]$ the completions are $C^0$, which give $J$ on
$[0,b_1]$. Suppose $J$ is known on $[0,b_n]$. Then:
\begin{itemize}[leftmargin=*,nosep]
\item $f=V_0+W$ is known on $[b_{n-1},b_n]$ and maps it onto $[b_n,b_{n+1}]$;
\item FIFO gives $C_k(f(u))=m^0_k+J_k(u)$ on $[b_n,b_{n+1}]$;
\item re-entry extends $J_{k+1}=r_kC_k$ to $[b_n,b_{n+1}]$.
\end{itemize}
Each step is forced by FIFO, re-entry, the arrivals and (F2); this proves uniqueness. For existence, two facts show that the
construction is a solution. First, $\sum_kc_kC_k(f(u))=V_0+W(u)=f(u)$, so the server works at unit rate and the constructed
$f$ equals $u+V(u)$. Second, $V(f(v))=W(f(v))-W(v)\ge\lambda c_1V(v)>0$, so $b_{n+1}>b_n$. Absolute continuity follows from
$C_k'=J_k'(u)/f'(u)\le J_k'(u)/(\lambda c_1)$. Moreover $V\ge\eta V_{\mathrm{tot}}$ with $\eta=\min_kc_k/R_k>0$, because both
are linear in $\pi\ge0$. By (iii), which needs only the unit work rate:
\begin{itemize}[leftmargin=*,nosep]
\item if $\rho\ge1$, then $V\ge\eta V_{\mathrm{tot}}(0)>0$ throughout, $b_{n+1}-b_n\ge\eta V_{\mathrm{tot}}(0)$, and
$b_n\to\infty$;
\item if $\rho<1$, then $b_n$ increases to the time at which $V_{\mathrm{tot}}$ vanishes (Theorem~\ref{thm:trap}(a)).
\end{itemize}

\emph{(iii).} With $j_k=J_k'$ and $\dot C_k=C_k'$, we have $\sum_kc_k\dot C_k=1$, $j_1=\lambda$ and $j_{k+1}=r_k\dot C_k$. Hence
\begin{align*}
V_{\mathrm{tot}}'&=\textstyle\sum_kR_k(j_k-\dot C_k)=\lambda R_1+\sum_k\dot C_k(r_kR_{k+1}-R_k)\\
&=\textstyle\lambda R_1-\sum_kc_k\dot C_k=\rho-1.
\end{align*}

\emph{(iv), (v).} Let $\Delta J_k(n)=J_k(u_n)-J_k(u_{n-1})$. By (ii), $\Delta_n=W(u_n)-W(u_{n-1})=\sum_kc_k\Delta J_k(n)$ for
$n\ge1$, and $\Delta_0=V(u_0)=\sum_kc_k\pi_k(u_0)$. For class 1, $\Delta J_1(n)=\lambda\Delta_{n-1}$. For $k\ge2$ and
$n\ge2$, FIFO gives $C_{k-1}(u_n)=m^0_{k-1}+J_{k-1}(u_{n-1})$ and the same at $u_{n-1}$, so
$\Delta J_k(n)=r_{k-1}\Delta J_{k-1}(n-1)$. For $n=1$, $C_{k-1}(u_1)-C_{k-1}(u_0)=\pi_{k-1}(u_0)$. Unrolling with
$\prod_{j=k-n}^{k-1}r_j=G_k/G_{k-n}$ gives (v) and then (iv).

The pending mass at $u_n$ is the joins in $(u_{n-1},u_n]$, because the completions by $u_n=f(u_{n-1})$ are the initial mass
plus the joins by $u_{n-1}$. For the sum, substitute $j=k-n$:
\[
\sum_nh_n=\sum_j\pi_j\sum_{n\ge0}c_{j+n}G_{j+n}/G_j=\sum_j\pi_jR_j=V_{\mathrm{tot}}.
\]
\end{proof}

\begin{lemma}[finite-memory renewal]\label{lem:renewal}
Let $A(s)=\sum_{k=1}^Kb_ks^k$ with $b_k\ge0$, $b_1>0$ and $A(1)=1$. Let
$y_n=\sum_{k=1}^{\min(n,K)}b_ky_{n-k}+g_n$, with $g_n\ge0$, $g_0>0$ and $g_n=0$ for $n\ge K$. Then $y_n=L+\epsilon_n$, where
$L=\sum_ng_n/\sum_kkb_k>0$ and $|\epsilon_n|\le C_r\vartheta^n$ for some $\vartheta<1$ that depends only on $(b_k)$, with
$C_r$ linear in $(g_n)$.
\end{lemma}

\begin{proof}
As formal power series, $\sum_ny_ns^n=g(s)/(1-A(s))$, where $g(s)=\sum_{n<K}g_ns^n$. The root $s=1$ of $1-A$ is simple, since
$A'(1)>0$. If $|s|<1$, then $|A(s)|\le A(|s|)<1$. If $|s|=1$ and $A(s)=1=\sum_kb_k$, then $s^k=1$ whenever $b_k>0$, in
particular $s=1$. So every other root lies outside the closed unit disk, at modulus at least $\vartheta_0^{-1}>1$. Fix
$\vartheta\in(\vartheta_0,1)$. Partial fractions give $g(s)/(1-A(s))=L/(1-s)+Q(s)$. Here $L=g(1)/A'(1)$ is the coefficient
of $1/(1-s)$, that is, minus the residue at $s=1$, and $Q$, a rational function plus a polynomial, is analytic for
$|s|<\vartheta_0^{-1}$. Cauchy's estimates on $|s|=\vartheta^{-1}$ finish the proof. This is the finite-memory case of the
discrete renewal theorem~\citep{feller1968introduction}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:trap}]
\emph{(a), $\rho<1$.} Take $u_0=0$. Summing the recursion, whose terms are all nonnegative, gives
$\sum_{n\le M}\Delta_n\le V_{\mathrm{tot}}(0)+\rho\sum_{n\le M}\Delta_n$, and in the limit
$\sum_n\Delta_n=V_{\mathrm{tot}}(0)/(1-\rho)$. Thus $b_n\uparrow\tau=V_{\mathrm{tot}}(0)/(1-\rho)$. For every orbit start
$u_0\in[0,V_0)$, each $\Delta_n(u_0)$ is positive, since $\Delta_n\ge a_1\Delta_{n-1}$. Every $t<\tau$ lies on such an
orbit, so the server is busy on $[0,\tau)$, and $V(b_n)=\Delta_n\to0$.

Without arrivals ($\lambda=0$, used for SHED), the orbit argument does not apply. There, $V_{\mathrm{tot}}'=-1$ and
$V\ge\eta V_{\mathrm{tot}}$ give $\tau=V_{\mathrm{tot}}(0)$ directly.

\emph{(b), $\rho=1$.} Lemma~\ref{lem:renewal} with $b_k=a_k$ gives $\Delta_n(u_0)\to V_{\mathrm{tot}}(u_0)/\sum_kka_k$,
uniformly in $u_0\in[0,V_0]$. Here $V_{\mathrm{tot}}(u_0)=V_{\mathrm{tot}}(0)$ and $\sum_kka_k=\bar k_c$. Every
$t\ge V_0$ equals $f^n(u_0)$ for a unique $u_0\in[0,V_0)$ and $n\ge1$. So
$V(t)\to V_{\mathrm{tot}}(0)/\bar k_c$, and by (v), $N\to\lambda E[T]V_{\mathrm{tot}}(0)/\bar k_c$.

\emph{(c), $\rho>1$.} Apply Lemma~\ref{lem:renewal} to $y_n=\Delta_nz^{-n}$, $b_k=a_kz^{-k}$ and $g_n=h_nz^{-n}$. It gives
$\Delta_n=z^n(L(u_0)+\epsilon_n)$ with $L$ continuous and positive on $[0,V_0]$ (since $h_0=V(u_0)>0$) and
$|\epsilon_n|\le C\vartheta^n$ uniformly. At $t=u_n(u_0)$,
\begin{gather*}
t=u_0+\textstyle\sum_{j<n}\Delta_j=\dfrac{Lz^n}{z-1}\bigl(1+o(1)\bigr),\\
V(t)=\Delta_n=Lz^n\bigl(1+o(1)\bigr),
\end{gather*}
so $V(t)/t\to z-1$ uniformly. The relative error is $O(n\max(\vartheta,1/z)^n)$. As $t\to\infty$, the generation index $n$
of $t$ tends to infinity because every $b_n$ is finite. By (v), $N(t)=\lambda\sum_kG_k\Delta_{n-k}$, and the identity
$(z-1)\sum_kG_kz^{-k}=1-E[z^{-T}]$ gives $N(t)/t\to\lambda(1-E[z^{-T}])$. The departure limit follows from
$\mathrm{Dep}=\sum_km_k^0+\lambda t-N$. By (v), the composition limit holds for the pending requests, and for the requests
completed over the next cycle, which are the same. Over a cycle $(t,f(t)]$, any per-turn weight $P_k$ is processed at the
average rate $\sum_kP_k\pi_k/\sum_kc_k\pi_k$.

\emph{(d), self-similarity.} $f(u)/u=1+V(u)/u\to z$, and $f^m(u)/u\to z^m$ by induction on $m$. A session that arrives at $u$
joins as turn~1 at $u$ and, with zero think time and FIFO, completes turn $m$ at $f^m(u)$. The centred form follows from the
virtual-origin argument below: $u_n+u^*=w_n$ with $u^*=V_{\mathrm{tot}}(0)/(\rho-1)$, and $w_{n+m}/w_n\to z^m$.

\emph{(e).} $F(x,\rho)=\rho\sum_k\omega_kx^{-k}-1$, with $\omega_k=c_kG_k/\sum_jc_jG_j$, satisfies $F(1,1)=0$ and
$\partial_xF(1,1)=-\bar k_c$. So $z'(1)=1/\bar k_c$. Then $E[z^{-T}]=1-(z-1)E[T]+O((z-1)^2)$, so
$\alpha=\lmiss(E[T]/\bar k_c)(\rho-1)+O((\rho-1)^2)$.

\emph{(d), continued: the virtual origin and the remaining work.} This sharpening was pointed out by an independent referee. For
$u\ge b_{K-1}$, all class-$k$ descendants of the initial requests have joined, and the arrivals' class-$k$ requests joined
by $u$ are those whose first request joined by $f^{-(k-1)}(u)$. Hence
\[
f(u)=V_{\mathrm{tot}}(0)+\sum_ka_kf^{-(k-1)}(u),
\]
and along every orbit $u_{n+1}=V_{\mathrm{tot}}(0)+\sum_ka_ku_{n+1-k}$. The constant $V_{\mathrm{tot}}(0)/(1-\rho)$ is a
particular solution, so $w_n=u_n-V_{\mathrm{tot}}(0)/(1-\rho)$ solves the homogeneous recursion. For $\rho<1$ this gives
$u_n\to\tau$ at once. For $\rho>1$, $w_n=Bz^n+O((r+\epsilon)^n)$, with $r$ the largest modulus among the other roots. Since
$V_{\mathrm{tot}}(u_n)=(\rho-1)w_n$,
\[
V(u_n)=\frac{z-1}{\rho-1}V_{\mathrm{tot}}(u_n)+O((r+\epsilon)^n),
\]
that is, $V(u_n)=\gamma\,(u_n+u^*)+O((r+\epsilon)^n)$.
In all eight cells, over $\rho\in[1.01,6.6]$, $r=0.946$--$0.966<1$, so the deviation vanishes absolutely. Correspondingly,
$f^m(u)+u^*\approx z^m(u+u^*)$. The transient decays by $\vartheta=r/z=0.84$--$0.95$ per
generation, and the registered window opens 33--37 generations after the burst. So we compare with the exact fluid as well
as with the limits.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:premium}]
\emph{The product.} Since $\lambda=1/\sum_kc_kG_kz^{-k}$, we have
$D/\lmiss=E[z^{-T}]\sum_kc_kG_k/\sum_kc_kG_kz^{-k}$. Multiplying and dividing by $\sum_kG_kz^{-k}$ and by
$E[T]=\sum_kG_k$ gives $H\Gamma$.

\emph{The factor $\Gamma$.} The ratio $\nu^z_k/\nu_k$ is proportional to $z^{-k}$, which is strictly decreasing, so $\nu^z$ is
below $\nu$ in the likelihood-ratio order. For non-decreasing $c$ this gives $E_{\nu^z}[c]\le E_\nu[c]$. By Abel summation,
\[
E_\nu[c]-E_{\nu^z}[c]=\sum_{k<K}\bigl(F_{\nu^z}(k)-F_\nu(k)\bigr)(c_{k+1}-c_k),
\]
in which every difference of distribution functions is positive. So equality holds only for constant $c$. The same argument
applied to $z'>z$ shows that $\Gamma$ is non-decreasing in $z$.

\emph{The factor $H$.} Since $\sum_kG_kz^{-k}=(1-E[z^{-T}])/(z-1)$, $H\ge1$ if and only if $E[z^{-T}](1+E[T](z-1))\ge1$.
For a geometric $T$ with mean $m$, $E[z^{-T}]=1/(1+m(z-1))$. For $T=\min(\mathrm{Geometric}(1-q),K)$ we have
$E[z^{-T}]=(1-q)\sum_kG_kz^{-k}+qG_Kz^{-K}$ and $(1-q)E[T]=1-q^K$. Hence
\[
H=1-q^K\Bigl(1-E[T]\Big/\sum_kG_kz^{K-k}\Bigr).
\]
For $z>1$ and $K\ge2$, the sum in the denominator exceeds $E[T]$ and increases with $z$. So $H$ lies in $(1-q^K,1)$ and
is non-increasing in~$z$.

\emph{Near the threshold.} The expansion follows from Theorem~\ref{thm:trap}(e), with
$\alpha=\lmiss(E[T]/\bar k_c)(\rho-1)+O((\rho-1)^2)$, and from $D=\lambda-\alpha$. When $\bar k_c=E[T]$, the sign of the
premium is decided at second order.

\emph{Monotonicity.} $\Gamma$ is strictly increasing for non-constant non-decreasing $c$ (by the Abel argument), and it
tends to $E_\nu[c]/c_1$ as $z\to\infty$. If that limit exceeds $1/(1-q^K)$, there is a unique $z_1$ with
$\Gamma(z_1)=1/(1-q^K)$. For $z\ge z_1$, $H\Gamma>(1-q^K)\Gamma(z_1)=1$. For $z\in[1+10^{-12},z_1]$ in our cells, the
monotonicity of $H$ and $\Gamma$ bounds the premium below on each bracket $[z_i,z_{i+1}]$ by $H(z_{i+1})\Gamma(z_i)-1$. That
bound is positive on a geometric grid of 53 brackets in all eight cells (\fpath{analysis/theory/icml/support.py},
evaluated in floating point).
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:trapthr}]
The state at the switching time $t_0$ is an admissible initial condition (F6), and $V(t_0)>0$. The claim then follows
from Theorem~\ref{thm:trap}(a)--(c) applied with rate $\lambda'$. If $\lambda'\in(\lmiss,D(\lambda))$, then $\rho'>1$, so
by Lemma~\ref{lem:orbit}(iii) $V_{\mathrm{tot}}(t)\ge V_{\mathrm{tot}}(t_0)$, and $V\ge\eta V_{\mathrm{tot}}(t_0)>0$:
the queue never empties, and it grows by Theorem~\ref{thm:trap}(c). At $\rho'=1$ the same bound holds, and the backlog tends
to $\lmiss E[T]V_{\mathrm{tot}}(t_0)/\bar k_c$.
\end{proof}

\noindent\emph{Proof of Corollary~\ref{cor:trapinv}.}

Divide Lemma~\ref{lem:perron}'s equation by $\sum_jc_jG_j$. The other quantities are explicit functions of $z$, $G$ and
$\omega$. Under $c\mapsto c/s$, $\omega$ is unchanged and $\lmiss\mapsto s\lmiss$. \qed

\paragraph{Remarks.}
\begin{itemize}[leftmargin=*,nosep]
\item \emph{Processor sharing.} The layer RR of \S\ref{sec:fp-heur} advances every session at $1/V(t)$ turns per second. Use
the clock $x=\int_0^t ds/V(s)$ and the continuous renewal theorem; the kernel is a step function with compact support, hence
non-lattice. This yields the Euler--Lotka equation $\lambda\sum_kc_kG_k\int_{k-1}^ke^{-\alpha_{\mathrm{PS}}a}\,da=1$ for the growth exponent
$\alpha_{\mathrm{PS}}$, the backlog growth $\lambda(1-E[e^{-\alpha_{\mathrm{PS}}T}])$, departures $\lambda E[e^{-\alpha_{\mathrm{PS}}T}]$
and the cycle slope $\alpha_{\mathrm{PS}}$. In overloaded processor sharing with service times $S$ the departures are
$\lambda E[e^{-\alpha_{\mathrm{PS}}S}]$, with $\alpha_{\mathrm{PS}}$ its own growth exponent, which can exceed $1/E[S]$: the
youth premium of Theorem~\ref{thm:premium} is the visit-indexed analogue of that property. This is the Crump--Mode--Jagers
structure of overloaded processor sharing~\citep{grishechkin1992processor,jeanmarie1994transient,puha2006overloaded}, with
reproduction rate $\lambda c_{\lfloor a\rfloor+1}$ at age~$a$; we do not write out the details. Near the threshold, the
backlog growth over $\lambda-\lmiss$ tends to $E[T]/(\bar k_c-\tfrac12)$, a half-turn offset from model~F. On the 300k cell,
$\alpha/(\lambda-\lmiss)$ is 0.647 and 0.666 under processor sharing at 1.35 and 1.65~$\lref$, against 0.634 and 0.653 for
model~F.
\item \emph{Stochastic version.} Take Poisson arrivals and any work-conserving order. The all-miss busy period started from
realized remaining works $w_i$ is the total work of a Galton--Watson tree, finite exactly when its progeny is. It ends with
probability $\exp(-\lambda(1-q)\sum_iw_i)$, where $q$ is the least root of $q=E[e^{-\lambda W(T)(1-q)}]$; the workload ever
falls to a level $w_b$ with probability $\exp(-\lambda(1-q)(\sum_iw_i-w_b))$~\citep{kendall1951some}. Take $w$ from the fluid
3000\,s after the registered burst on the 300k cell (\fpath{support.py}, \texttt{kendall\_numbers}). Emptying then has
probability 0.35 at $\rho=1.01$, 0.12 at 1.02, $4\times10^{-3}$ at 1.05, $10^{-5}$ at 1.10, $4\times10^{-23}$ at 1.35 and
$2\times10^{-54}$ at 1.65. These are orders of magnitude (a realized $w$ fluctuates around the fluid value), and hits return
before the queue empties, so they are not bounds on a spontaneous exit; they say it matters only within a few percent of
$\lmiss$. The branching description is classical and claimed as such.
\item \emph{Steady solutions of FIFO re-entrant lines.} Model~F is a single-station re-entrant line~\citep{kumar1993reentrant}
in overload. Khanin et al.~\citep{khanin2001steady} study steady (constant-flow) solutions of fluid FIFO re-entrant lines from the
empty state, with stage-dependent service times. At an overloaded station the steady flows satisfy
$x_s=x_{s-1}/\sum_jx_{j-1}v_j$; at a single station with $n$ fixed stages this gives $x_s=a^s$ with $\sum_jv_ja^j=1$, which is
Lemma~\ref{lem:perron}'s equation with $a=1/z$ and $T\equiv n$, and the steady flows give part~(c)'s gap slope, composition and
departure rate; a closing remark extends their uniqueness theorem to a fixed fraction of the flow leaving at each node, which in
a fluid is our random $T$, and their normalization gives Corollary~\ref{cor:trapinv}. They prove existence, uniqueness for
Kelly-type lines and non-uniqueness examples for two stations, and leave the uniqueness of general (non-steady) trajectories open.
Relative to theirs, our contribution is the renewal recursion along FIFO orbits from an arbitrary initial state, hence the
convergence of every trapped trajectory to the steady regime with an exact virtual origin and rate, and the turn-count and
cost-growth factors of Theorem~\ref{thm:premium}; Corollary~\ref{cor:trapthr} is their operational reading with the classical drain
condition. The critical case,
Theorem~\ref{thm:trap}(b), is the fluid counterpart of heavy-traffic state-space collapse in multiclass FIFO feedback
queues~\citep{reiman1988multiclass,dai1995multiclass}. Overloaded multiclass FIFO queues with abandonment and no re-entry have fluid
limits~\citep{jennings2013fluid}; fluid metastability of queues with load-dependent service is posed by
Dong~\citep{dong2022metastability}; and within-request KV growth under continuous batching is analysed by Ao et
al.~\citep{ao2026service}, a different instability.
\item \emph{Think time.} With a 5\,s think time (the workload's mean tool time) the exact fluid's statistics over the registered
window agree with zero think time to within 0.02\% (backlog slope, completion rate, mean served prompt); the mean cycle
is 5\,s shorter and its slope the same.
\item \emph{How fast the limits are reached.} The transient shrinks by $\vartheta=r/z=0.84$--$0.95$ per generation over
$\rho\in[1.01,6.6]$. At 300k the registered window starts about 33 (1.65~$\lref$) or 37 (1.35~$\lref$) generations after the
burst, so a transient of order $\vartheta^{35}\approx0.1$ remains; the asymptotic values and the exact fluid over the window
differ by about 1\% for the composition and by 4--12\% for the backlog slope. No general rate is claimed: a nearly periodic kernel
converges slowly (with $T=4$ always, the fluid is still 3\% off after 1000 generations).
\end{itemize}
